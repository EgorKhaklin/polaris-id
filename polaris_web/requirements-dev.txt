# polaris_web/requirements-dev.txt: run AND test, in one install.
#
# The runtime lock (what the images install) and the test lock, both hashed, so
# pip checks every file it installs. The images never install this file.
#
#   python3.12 -m venv polaris_web/venv
#   source polaris_web/venv/bin/activate
#   pip install -r polaris_web/requirements-dev.txt
#
# CI installs this file; pip-audit gates on requirements.txt (the image surface)
# and audits this one informationally. To change a version, edit the .in file
# and recompile its .txt (see the header of each .in).

-r requirements.txt
-r requirements-test.txt
