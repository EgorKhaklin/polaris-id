% v2-figures/offline-status.tex: the freshness trade-off of the signed status
% assertion, drawn as a timeline. The shaded window is the honest cost of
% offline authorization: a revoked credential's last ACTIVE assertion stays
% valid until it expires, bounded by the TTL the issuer set and the max_age
% the verifier accepts.
\begin{tikzpicture}
  % axis
  \draw[flow, line width=1.2pt] (0,0) -- (14.2,0) node[anchor=west, font=\scriptsize\color{steelgray}] {время};
  % the assertion's window
  \fill[navymid!12!white] (1.0,-0.28) rectangle (9.0,0.28);
  \draw[navymid, line width=0.8pt] (1.0,-0.28) rectangle (9.0,0.28);
  % the revoked-but-valid window
  \fill[classmark!15!white] (6.2,-0.28) rectangle (9.0,0.28);
  \draw[classmark, line width=0.8pt, dash pattern=on 2pt off 2pt] (6.2,-0.28) rectangle (9.0,0.28);
  % events
  \foreach \x in {1.0, 3.0, 4.6, 6.2, 9.0, 11.4} { \draw[navydark, line width=1pt] (\x,-0.15) -- (\x,0.15); }
  \node[font=\scriptsize, anchor=south, align=center] at (1.0,1.55) {эмитент подписывает\\\code{состояние = ДЕЙСТВУЮЩИЙ}};
  \node[font=\scriptsize, anchor=north, align=center] at (3.0,-0.45) {держатель\\подшивает его\\к предъявлению};
  \node[font=\scriptsize, anchor=south, align=center] at (4.6,1.55) {доверяющая сторона А,\\автономно: принять};
  \node[font=\scriptsize\bfseries\color{classmark}, anchor=north, align=center] at (6.2,-0.45) {эмитент отзывает\\токен};
  \node[font=\scriptsize, anchor=south, align=center] at (9.0,1.55) {утверждение истекает;\\теперь можно вычеканить\\только \code{ОТОЗВАН}};
  \node[font=\scriptsize, anchor=north, align=center] at (11.4,-0.45) {доверяющая сторона Б,\\автономно: отвергнуть};
  \foreach \x/\y in {1.0/1.5, 4.6/1.5, 9.0/1.5} { \draw[steelgray, line width=0.4pt, opacity=0.6] (\x,0.3) -- (\x,\y); }

  % the window's meaning, below the event labels
  \draw[classmark, line width=0.6pt, dash pattern=on 2pt off 2pt] (6.2,-1.35) -- (6.2,-1.9);
  \draw[classmark, line width=0.6pt, dash pattern=on 2pt off 2pt] (9.0,-0.32) -- (9.0,-1.9);

  % the two remedies
  \node[softbox, anchor=north east, text width=6.3cm, minimum height=1.75cm] at (6.8,-3.7) {\textbf{Как сузить:} верификатор навязывает собственный более короткий \code{макс\_возраст}; сторона с высоким риском требует секунд либо выходит в сеть.};
  \node[softbox, anchor=north west, text width=6.3cm, minimum height=1.75cm] at (7.4,-3.7) {\textbf{Размен:} ненаблюдаемость эмитентом, автономную доступность и свежесть отзыва нельзя довести до предела разом. Окно есть число политики, а не закон.};
\end{tikzpicture}
