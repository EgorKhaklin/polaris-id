% v3-figures/lifecycle.tex: the token lifecycle state machine, drawn from the trigger that
% enforces it (06_triggers.sql, enforce_token_state_machine) rather than from Version 1's
% diagram. The six legal transitions, each labelled on its own arrow with the event type
% audit_token_state_change writes for it: RESERVE->ACTIVE (ACTIVATED), ACTIVE->DORMANT
% (DEACTIVATED), ACTIVE->EXPIRED, ACTIVE->LOST, ACTIVE->REVOKED, and RESERVE->REVOKED, the
% administrative revocation of a token never activated, which Version 1 left out of its
% figure although the trigger has always allowed it. check_token_core_triggers_are_bound
% holds the trigger to exactly these six.
\begin{tikzpicture}[
  scale=0.78, font=\small,
  state/.style={draw=navymid, line width=1pt, rounded corners=2pt, minimum width=2.3cm, minimum height=0.9cm,
    fill=navydark, text=white, font=\bfseries\small, align=center},
  centerstate/.style={draw=accent, line width=1.2pt, rounded corners=2pt, minimum width=3.1cm, minimum height=0.9cm,
    fill=accent, text=white, font=\bfseries\small, align=center},
  termstate/.style={draw=steelgray, line width=1pt, rounded corners=2pt, minimum width=2.3cm, minimum height=0.9cm,
    fill=steelgray!85, text=white, font=\bfseries\small, align=center},
  trans/.style={->, draw=navymid, line width=1pt, >={Stealth[length=2.5mm]}},
  termtrans/.style={->, draw=accent, line width=1pt, >={Stealth[length=2.5mm]}, dash pattern=on 4pt off 2.5pt},
  lbl/.style={font=\scriptsize\itshape, text=navydark, inner sep=2pt, fill=white, fill opacity=0.9, text opacity=1},
]
\node[state]       (reserve) at (0,4.3)     {RESERVE};
\node[centerstate] (active)  at (0,1.3)     {ACTIVE};
\node[state]       (dormant) at (7.2,1.3)   {DORMANT};
\node[termstate]   (expired) at (-5.6,-2.1) {EXPIRED};
\node[termstate]   (lost)    at (0,-2.1)    {LOST};
\node[termstate]   (revoked) at (5.6,-2.1)  {REVOKED};

\draw[trans] ($(reserve.north)+(0,1.1)$) -- (reserve.north) node[lbl, midway, right=1pt] {ISSUED};
\draw[trans] (reserve) -- (active) node[lbl, midway, right=1pt] {ACTIVATED};
\draw[trans] (active) -- (dormant) node[lbl, midway, above=1pt] {DEACTIVATED};
\draw[termtrans] (active) -- (expired) node[lbl, midway, sloped, above=1pt] {EXPIRED};
\draw[termtrans] (active) -- (lost) node[lbl, midway, right=1pt] {LOST};
\draw[termtrans] (active) -- (revoked) node[lbl, midway, sloped, above=1pt] {REVOKED};
\draw[termtrans, rounded corners=10pt] (reserve.east) -- (10.0,4.3) -- (10.0,-2.1) -- (revoked.east);
\node[lbl, above=1pt] at (6.2,4.3) {REVOKED, never activated};

\draw[trans] (-3.9,-3.55) -- (-2.9,-3.55);
\node[lbl, anchor=west] at (-2.8,-3.55) {to an operational state};
\draw[termtrans] (1.4,-3.55) -- (2.4,-3.55);
\node[lbl, anchor=west] at (2.5,-3.55) {to a terminal state};
\end{tikzpicture}
